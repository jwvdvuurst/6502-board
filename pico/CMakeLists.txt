cmake_minimum_required(VERSION 3.20)
option(BRIDGE_HOST_TESTS "Build portable protocol tests without Pico SDK" OFF)
if(BRIDGE_HOST_TESTS)
    project(bridge_tests C)
    enable_testing()
    add_executable(bridge_tests tests/test_bridge.c bridge.c terminal.c)
    target_include_directories(bridge_tests PRIVATE .)
    target_compile_options(bridge_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME bridge_tests COMMAND bridge_tests)
    return()
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../firmware/tools/pico-sdk")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(board_bridge C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(board_bridge main.c bridge.c terminal.c usb_descriptors.c)
target_include_directories(board_bridge PRIVATE .)
target_compile_options(board_bridge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(board_bridge pico_stdlib pico_unique_id tinyusb_device hardware_gpio)
pico_enable_stdio_usb(board_bridge 0)
pico_enable_stdio_uart(board_bridge 0)
pico_add_extra_outputs(board_bridge)
